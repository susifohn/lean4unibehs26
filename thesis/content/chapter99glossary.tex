\section*{Symbols}
\begin{longtable}{@{}p{0.2\textwidth}p{0.75\textwidth}@{}}
  $\Gamma$             & context (set of assumptions) \\
  $\Gamma \vdash A$    & judgement: $A$ is derivable from $\Gamma$ \\
  $A[t/x]$             & substitution of $t$ for $x$ in $A$ \\
  $A \to B$            & implication / function type \\
  % $...$              & ... \\   % <- add your own here
\end{longtable}

\section*{Definitions}
\begin{longtable}{@{}p{0.2\textwidth}p{0.75\textwidth}@{}}
  \textbf{Derivation}  & finite tree of rule applications ending in a conclusion. \\
  \textbf{Judgement}   & statement of the form $\Gamma \vdash A$. \\
  \textbf{Deduction}   & reasoning from premises to a conclusion. \\
